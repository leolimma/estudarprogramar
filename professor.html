Script SQL das Tabelas (Execute no SQL Editor do Supabase): 
-- 1. Tabela de Tentativas dos Alunos
CREATE TABLE IF NOT EXISTS student_attempts (
    id TEXT PRIMARY KEY,
    student_name TEXT NOT NULL,
    student_class TEXT NOT NULL,
    student_login TEXT,
    character_name TEXT,
    character_avatar TEXT,
    score INTEGER DEFAULT 0,
    progress_percentage INTEGER DEFAULT 0,
    stages_completed INTEGER DEFAULT 0,
    correct_answers INTEGER DEFAULT 0,
    wrong_attempts INTEGER DEFAULT 0,
    lives_remaining INTEGER DEFAULT 3,
    status TEXT,
    stage_results JSONB,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 2. Tabela de Contas de Alunos
CREATE TABLE IF NOT EXISTS student_users (
    id TEXT PRIMARY KEY,
    login TEXT UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    first_name TEXT NOT NULL,
    student_class TEXT NOT NULL,
    avatar TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Desabilitar RLS para sala de aula sem bloqueio
ALTER TABLE student_attempts DISABLE ROW LEVEL SECURITY;
ALTER TABLE student_users DISABLE ROW LEVEL SECURITY;
